const express = require("express"); const multer = require("multer"); const fs = require("fs"); const path = require("path"); const crypto = require("crypto"); const app = express(); const PORT = process.env.PORT || 3000; const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || "Admin2137!"; const DATA_DIR = path.join(__dirname, "data"); const UPLOADS_DIR = path.join(__dirname, "uploads"); const ORDERS_FILE = path.join(DATA_DIR, "orders.json"); const BALANCE_FILE = path.join(DATA_DIR, "balance.json"); // ===================================================== // FOLDERS // ===================================================== if (!fs.existsSync(DATA_DIR)) { fs.mkdirSync(DATA_DIR, { recursive: true }); } if (!fs.existsSync(UPLOADS_DIR)) { fs.mkdirSync(UPLOADS_DIR, { recursive: true }); } // ===================================================== // DEFAULT FILES // ===================================================== if (!fs.existsSync(ORDERS_FILE)) { fs.writeFileSync( ORDERS_FILE, JSON.stringify([], null, 2), "utf8" ); } if (!fs.existsSync(BALANCE_FILE)) { fs.writeFileSync( BALANCE_FILE, JSON.stringify( { balance: 0, maxBalance: 100 }, null, 2 ), "utf8" ); } // ===================================================== // MIDDLEWARE // ===================================================== app.use(express.json({ limit: "10mb" })); app.use(express.urlencoded({ extended: true })); app.use(express.static(__dirname)); app.use( "/uploads", express.static(UPLOADS_DIR) ); // ===================================================== // MULTER // ===================================================== const storage = multer.diskStorage({ destination: function (req, file, cb) { cb(null, UPLOADS_DIR); }, filename: function (req, file, cb) { const ext = path.extname(file.originalname); const name = Date.now() + "-" + crypto.randomBytes(5).toString("hex") + ext; cb(null, name); } }); const upload = multer({ storage: storage, limits: { fileSize: 25 * 1024 * 1024 }, fileFilter: function (req, file, cb) { const allowed = [ ".stl", ".obj", ".3mf" ]; const ext = path.extname(file.originalname) .toLowerCase(); if (!allowed.includes(ext)) { return cb( new Error( "Dozwolone są tylko pliki STL, OBJ i 3MF." ) ); } cb(null, true); } }); // ===================================================== // HELPERS // ===================================================== function readOrders() { try { const data = fs.readFileSync( ORDERS_FILE, "utf8" ); const orders = JSON.parse(data); return Array.isArray(orders) ? orders : []; } catch (error) { console.error( "Błąd odczytu orders.json:", error ); return []; } } function saveOrders(orders) { fs.writeFileSync( ORDERS_FILE, JSON.stringify( orders, null, 2 ), "utf8" ); } function readBalance() { try { const data = fs.readFileSync( BALANCE_FILE, "utf8" ); const balance = JSON.parse(data); return { balance: Number(balance.balance) || 0, maxBalance: Number(balance.maxBalance) || 100 }; } catch (error) { return { balance: 0, maxBalance: 100 }; } } function saveBalance(balance) { fs.writeFileSync( BALANCE_FILE, JSON.stringify( balance, null, 2 ), "utf8" ); } function generateId(prefix) { return ( prefix + "-" + Date.now() + "-" + crypto .randomBytes(4) .toString("hex") ); } function clean(value) { if ( value === undefined || value === null ) { return ""; } return String(value).trim(); } function adminAuth(req, res, next) { const password = req.headers["x-admin-password"]; if (password !== ADMIN_PASSWORD) { return res.status(401).json({ error: "Nieprawidłowe hasło administratora." }); } next(); } // ===================================================== // HOME // ===================================================== app.get("/", function (req, res) { res.sendFile( path.join( __dirname, "index.html" ) ); }); app.get("/admin", function (req, res) { res.sendFile( path.join( __dirname, "admin.html" ) ); }); // ===================================================== // ADMIN LOGIN // ===================================================== app.post( "/api/admin/login", function (req, res) { const password = clean(req.body.password); if ( password !== ADMIN_PASSWORD ) { return res.status(401).json({ success: false, error: "Złe hasło." }); } res.json({ success: true, message: "Zalogowano." }); } ); // ===================================================== // ADMIN BALANCE // ===================================================== app.get( "/api/admin/balance", adminAuth, function (req, res) { const data = readBalance(); res.json({ success: true, balance: Number( data.balance.toFixed(2) ), maxBalance: data.maxBalance }); } ); // ===================================================== // ADMIN ORDERS // ===================================================== app.get( "/api/admin/orders", adminAuth, function (req, res) { const orders = readOrders(); res.json({ success: true, orders: orders.reverse() }); } ); // ===================================================== // CREATE NORMAL ORDER // ===================================================== app.post( "/api/orders", function (req, res) { try { const body = req.body || {}; const name = clean(body.name); const email = clean(body.email); const address = clean(body.address); const items = Array.isArray(body.items) ? body.items : []; let subtotal = Number(body.subtotal) || 0; let delivery = Number(body.delivery) || 0; let discount = Number(body.discount) || 0; let total = Number(body.total) || 0; if (!name) { return res.status(400).json({ error: "Brak imienia." }); } if (!email) { return res.status(400).json({ error: "Brak adresu e-mail." }); } if (!address) { return res.status(400).json({ error: "Brak adresu dostawy." }); } if (!items.length) { return res.status(400).json({ error: "Koszyk jest pusty." }); } // Bezpieczne ograniczenia subtotal = Math.max( 0, Math.min( subtotal, 100000 ) ); delivery = Math.max( 0, Math.min( delivery, 1000 ) ); discount = Math.max( 0, Math.min( discount, subtotal ) ); total = Math.max( 0, Math.min( total, 100000 ) ); const order = { id: generateId("ORDER"), type: "normal", name, email, address, phone: clean(body.phone), items, subtotal, delivery, discount, discountCode: clean( body.discountCode ) || null, total, status: "NEW", createdAt: new Date().toISOString() }; const orders = readOrders(); orders.push(order); saveOrders(orders); // ========================================== // DODAWANIE DO SALDA // Maksymalnie 100 zł // ========================================== const balanceData = readBalance(); const oldBalance = Number(balanceData.balance) || 0; const maxBalance = 100; const newBalance = Math.min( maxBalance, oldBalance + total ); balanceData.balance = Number( newBalance.toFixed(2) ); balanceData.maxBalance = maxBalance; saveBalance(balanceData); console.log( "[ORDER]", order.id, "|", total.toFixed(2), "PLN", "| SALDO:", balanceData.balance.toFixed(2), "PLN" ); res.status(201).json({ success: true, orderId: order.id, message: "Zamówienie zostało wysłane.", balance: balanceData.balance }); } catch (error) { console.error( "ORDER ERROR:", error ); res.status(500).json({ error: "Nie udało się utworzyć zamówienia." }); } } ); // ===================================================== // CUSTOM MODEL ORDER // ===================================================== app.post( "/api/orders/custom", upload.single("file"), function (req, res) { try { const name = clean(req.body.name); const email = clean(req.body.email); const phone = clean(req.body.phone); const material = clean(req.body.material); const color = clean(req.body.color); const description = clean( req.body.description ); if (!name) { if (req.file) { fs.unlinkSync( req.file.path ); } return res.status(400).json({ error: "Podaj imię." }); } if (!email) { if (req.file) { fs.unlinkSync( req.file.path ); } return res.status(400).json({ error: "Podaj e-mail." }); } if (!description) { if (req.file) { fs.unlinkSync( req.file.path ); } return res.status(400).json({ error: "Opisz model." }); } if (!req.file) { return res.status(400).json({ error: "Nie wysłano pliku." }); } const order = { id: generateId("CUSTOM"), type: "custom", name, email, phone, material, color, description, file: { originalName: req.file.originalname, filename: req.file.filename, size: req.file.size, url: "/uploads/" + req.file.filename }, price: null, total: 0, status: "WAITING_QUOTE", createdAt: new Date().toISOString() }; const orders = readOrders(); orders.push(order); saveOrders(orders); console.log( "[CUSTOM MODEL]", order.id, "|", order.file.originalName ); res.status(201).json({ success: true, orderId: order.id, message: "Model został wysłany do administratora." }); } catch (error) { console.error( "CUSTOM ERROR:", error ); if ( req.file && fs.existsSync(req.file.path) ) { fs.unlinkSync( req.file.path ); } res.status(500).json({ error: error.message || "Nie udało się wysłać modelu." }); } } ); // ===================================================== // ADMIN — CHANGE ORDER STATUS // ===================================================== app.patch( "/api/admin/orders/:id", adminAuth, function (req, res) { const id = req.params.id; const status = clean(req.body.status); const allowedStatuses = [ "NEW", "WAITING_QUOTE", "PAID", "IN_PROGRESS", "READY", "SENT", "DONE", "CANCELLED" ]; if ( !allowedStatuses.includes( status ) ) { return res.status(400).json({ error: "Nieprawidłowy status." }); } const orders = readOrders(); const index = orders.findIndex( order => order.id === id ); if (index === -1) { return res.status(404).json({ error: "Nie znaleziono zamówienia." }); } orders[index].status = status; orders[index].updatedAt = new Date().toISOString(); saveOrders(orders); res.json({ success: true, order: orders[index] }); } ); // ===================================================== // ADMIN — SET PRICE FOR CUSTOM MODEL // ===================================================== app.patch( "/api/admin/orders/:id/price", adminAuth, function (req, res) { const id = req.params.id; const price = Number(req.body.price); if ( !Number.isFinite(price) || price < 0 || price > 10000 ) { return res.status(400).json({ error: "Cena musi być od 0 do 10000 zł." }); } const orders = readOrders(); const index = orders.findIndex( order => order.id === id ); if (index === -1) { return res.status(404).json({ error: "Nie znaleziono zamówienia." }); } const oldTotal = Number( orders[index].total ) || 0; orders[index].price = Number( price.toFixed(2) ); orders[index].total = Number( price.toFixed(2) ); orders[index].status = "WAITING_PAYMENT"; orders[index].updatedAt = new Date().toISOString(); saveOrders(orders); res.json({ success: true, order: orders[index], oldTotal, newTotal: orders[index].total }); } ); // ===================================================== // ADMIN — MARK CUSTOM ORDER AS PAID // ===================================================== app.post( "/api/admin/orders/:id/pay", adminAuth, function (req, res) { const id = req.params.id; const orders = readOrders(); const index = orders.findIndex( order => order.id === id ); if (index === -1) { return res.status(404).json({ error: "Nie znaleziono zamówienia." }); } const order = orders[index]; const amount = Number(order.total) || 0; if (amount <= 0) { return res.status(400).json({ error: "To zamówienie nie ma ustawionej ceny." }); } if (order.paid) { return res.status(400).json({ error: "To zamówienie jest już opłacone." }); } order.paid = true; order.status = "PAID"; order.paidAt = new Date().toISOString(); // ========================================== // SALDO MAKSYMALNIE 100 ZŁ // ========================================== const balanceData = readBalance(); const oldBalance = Number(balanceData.balance) || 0; const maxBalance = 100; const newBalance = Math.min( maxBalance, oldBalance + amount ); balanceData.balance = Number( newBalance.toFixed(2) ); balanceData.maxBalance = maxBalance; saveBalance( balanceData ); saveOrders( orders ); console.log( "[PAID]", order.id, "| +", amount.toFixed(2), "PLN", "| SALDO:", balanceData.balance.toFixed(2), "PLN" ); res.json({ success: true, message: "Zamówienie oznaczone jako opłacone.", order, balance: balanceData.balance }); } ); // ===================================================== // ADMIN — DELETE ORDER // ===================================================== app.delete( "/api/admin/orders/:id", adminAuth, function (req, res) { const id = req.params.id; let orders = readOrders(); const order = orders.find( item => item.id === id ); if (!order) { return res.status(404).json({ error: "Nie znaleziono zamówienia." }); } // Usuń plik własnego modelu if ( order.file && order.file.filename ) { const filePath = path.join( UPLOADS_DIR, order.file.filename ); if ( fs.existsSync(filePath) ) { fs.unlinkSync( filePath ); } } orders = orders.filter( item => item.id !== id ); saveOrders(orders); res.json({ success: true, message: "Zamówienie usunięte." }); } ); // ===================================================== // ADMIN — RESET BALANCE // ===================================================== app.post( "/api/admin/balance/reset", adminAuth, function (req, res) { const balanceData = readBalance(); balanceData.balance = 0; balanceData.maxBalance = 100; saveBalance( balanceData ); res.json({ success: true, balance: 0, message: "Saldo zostało wyzerowane." }); } ); // ===================================================== // ADMIN — WITHDRAW // ===================================================== app.post( "/api/admin/balance/withdraw", adminAuth, function (req, res) { const amount = Number(req.body.amount); if ( !Number.isFinite(amount) || amount <= 0 ) { return res.status(400).json({ error: "Podaj prawidłową kwotę." }); } const balanceData = readBalance(); if ( amount > balanceData.balance ) { return res.status(400).json({ error: "Nie masz tyle środków." }); } balanceData.balance = Number( ( balanceData.balance - amount ).toFixed(2) ); saveBalance( balanceData ); console.log( "[WITHDRAW]", amount.toFixed(2), "PLN", "| SALDO:", balanceData.balance.toFixed(2), "PLN" ); res.json({ success: true, withdrawn: amount, balance: balanceData.balance }); } ); // ===================================================== // 404 API // ===================================================== app.use( "/api", function (req, res) { res.status(404).json({ error: "Nie znaleziono endpointu API." }); } ); // ===================================================== // ERROR HANDLER // ===================================================== app.use( function (error, req, res, next) { console.error( "SERVER ERROR:", error ); if ( error instanceof multer.MulterError ) { if ( error.code === "LIMIT_FILE_SIZE" ) { return res.status(400).json({ error: "Plik jest za duży. Maksymalnie 25 MB." }); } } res.status(500).json({ error: error.message || "Wystąpił błąd serwera." }); } ); // ===================================================== // START SERVER // ===================================================== app.listen( PORT, "0.0.0.0", function () { console.log(""); console.log( "========================================" ); console.log( " PRINTERLASE3D SERVER" ); console.log( "========================================" ); console.log( "Port:", PORT ); console.log( "Sklep działa." ); console.log( "Admin:", "/admin.html" ); console.log( "Saldo maksymalne:", "100 PLN" ); console.log( "========================================" ); console.log(""); } );
